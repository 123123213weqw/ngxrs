# Declared conformance list.
#
# Scope: what the reference build can actually do, not "an nginx default".
# The baseline is configured (see BENCH.md) with
#   --with-http_realip_module --with-http_auth_request_module
#   --with-http_slice_module --with-http_sub_module
#   --with-http_secure_link_module --with-http_dav_module
#   --with-http_stub_status_module --with-http_gzip_static_module
#   --with-threads --with-stream --with-stream_ssl_module
#   --with-stream_realip_module --with-mail --with-mail_ssl_module
# so those modules are IN scope; anything the reference cannot be built to
# expose (lua, njs, perl, xslt, image_filter, mp4, geoip, ...) is out.
#
# Selection rule, applied mechanically from a full 348-file scan:
#   1. the suite is inside the scope above;
#   2. BOTH servers emitted at least one assertion (a suite that skips on both
#      sides proves nothing and is excluded -- see the SKIP bucket);
#   3. no assertion passes on nginx and fails on ngxrs (WORSE = 0);
#   4. the run is not truncated on our side (TRUNC = 0), i.e. both sides emitted
#      the same number of assertion lines.
# Files where BOTH servers fail some assertions are allowed and kept: the
# criterion is "nginx passes and ngxrs does not", not absolute conformance.
#
# Counts are the last measured ngxrs result.

# ---- realip ----
# `set_real_ip_from` accepts an address, a CIDR, a hostname resolved at
# configuration time, and `unix:`; the peer must be trusted before its header
# is believed, which decides `$remote_addr` and therefore `allow`/`deny`,
# the limit_req/limit_conn keys and access_log.
realip.t
realip_hostname.t
realip_remote_addr.t
realip_remote_port.t

# ---- rewrite / if ----
# `if` operators: truthiness, `=`/`!=`, `~`/`~*`/`!~`/`!~*`, and the file
# tests (`-f -d -e -x` and their `!` forms), whose operand is a path with
# variables in it.
rewrite_if.t

# ---- stream geo / map ----
# `geo` (CIDR and `ranges`, `default`, `delete`, `include`) and `map` in
# `stream {}` context, plus repeated `listen` in one stream server.
stream_geo.t
stream_map.t

# ---- resolver ----
# A + AAAA in parallel, CNAME following with nginx's `process_a` state machine,
# TTL/`valid=` caching, per-name server rotation, resend, and TCP fallback on a
# truncated answer.
http_resolver.t
http_resolver_aaaa.t

# ---- autoindex ----
# `autoindex_format html|json|jsonp|xml`. nginx's JSON/XML bodies are compared
# byte for byte, including the fact that entry names are emitted as raw UTF-8
# bytes and only `\\`, `"` and control characters are escaped.
autoindex_format.t

# ---- PROXY protocol ----
# `listen ... proxy_protocol` (v1 text and v2 binary with TLVs),
# `$proxy_protocol_*`, `real_ip_header proxy_protocol`, and `stream`'s
# `proxy_protocol on`. `allow`/`deny` and the access log follow the rewritten
# address because nginx's realip runs in POST_READ, before ACCESS.
proxy_protocol.t
proxy_protocol2.t
proxy_protocol2_tlv.t
proxy_protocol_ipv6.t

# ---- dav ----
# `dav_methods PUT DELETE MKCOL COPY MOVE`, `dav_access`,
# `create_full_put_path`, `min_delete_depth`, and the `Depth`/`Overwrite`/
# `Destination` headers.
dav.t
dav_chunked.t
dav_utf8.t

# ---- secure_link ----
# Both styles: `secure_link`+`secure_link_md5` (base64url MD5, optional
# expiry) and the old `secure_link_secret` path form, including inheritance.
secure_link.t

# ---- sub_filter ----
# Multiple patterns, longest match first, per-pattern `once`, variables in both
# operands, and the filter running on proxied responses as well as local ones.
