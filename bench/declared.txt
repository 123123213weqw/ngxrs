# Declared conformance list.
#
# Scope: what the reference build can actually do, not "an nginx default".
# The baseline is configured (see BENCH.md) with
#   --with-http_realip_module --with-http_auth_request_module
#   --with-http_slice_module --with-http_sub_module
#   --with-http_secure_link_module --with-http_dav_module
#   --with-http_stub_status_module --with-http_gzip_static_module
#   --with-threads --with-stream --with-stream_ssl_module
#   --with-stream_realip_module --with-mail --with-mail_ssl_module
# so those modules are IN scope; anything the reference cannot be built to
# expose (lua, njs, perl, xslt, image_filter, mp4, geoip, ...) is out.
#
# Selection rule, applied mechanically from a full 348-file scan:
#   1. the suite is inside the scope above;
#   2. BOTH servers emitted at least one assertion (a suite that skips on both
#      sides proves nothing and is excluded -- see the SKIP bucket);
#   3. no assertion passes on nginx and fails on ngxrs (WORSE = 0);
#   4. the run is not truncated on our side (TRUNC = 0), i.e. both sides emitted
#      the same number of assertion lines.
# Files where BOTH servers fail some assertions are allowed and kept: the
# criterion is "nginx passes and ngxrs does not", not absolute conformance.
#
# Counts are the last measured ngxrs result.

# ---- realip ----
# `set_real_ip_from` accepts an address, a CIDR, a hostname resolved at
# configuration time, and `unix:`; the peer must be trusted before its header
# is believed, which decides `$remote_addr` and therefore `allow`/`deny`,
# the limit_req/limit_conn keys and access_log.
realip.t
realip_hostname.t
realip_remote_addr.t
realip_remote_port.t

# ---- rewrite / if ----
# `if` operators: truthiness, `=`/`!=`, `~`/`~*`/`!~`/`!~*`, and the file
# tests (`-f -d -e -x` and their `!` forms), whose operand is a path with
# variables in it.
rewrite_if.t

# ---- stream PROXY protocol & TLS variables ----
# `listen ... proxy_protocol` on the stream side (input), `$proxy_protocol_*`
# including TLVs, the header written before a TLS handshake when `proxy_ssl` and
# `proxy_protocol` are both on, and the `$ssl_*` facts a terminated connection
# reports.
stream_proxy_protocol.t
stream_proxy_protocol_ssl.t
stream_proxy_protocol_ipv6.t
stream_proxy_protocol2_tlv.t
stream_ssl_variables.t
stream_proxy_ssl_name_complex.t

# ---- stream geo / map ----
# `geo` (CIDR and `ranges`, `default`, `delete`, `include`) and `map` in
# `stream {}` context, plus repeated `listen` in one stream server.
stream_geo.t
stream_map.t
stream_set.t
stream_split_clients.t

# ---- request_id / access_log ----
# `$request_id` (one 32-hex id per request, shared with subrequests and the log),
# and the log writer's details: longest-match variable expansion, `${name}`,
# `if=`, per-format `escape=`, dynamic log paths, and `-` for a not-found
# variable in the default escape mode.
request_id.t
# The log writer end to end: `if=`, per-format `escape=`, dynamic log paths,
# and `open_log_file_cache` (a handle cached within `valid` keeps writing to the
# inode after the path is renamed away; `min_uses` and `max` decide which paths
# get that treatment).
access_log.t
access_log_variables.t

# ---- HTTP/2 header filter ----
# `absolute_redirect` / `server_name_in_redirect` / `port_in_redirect` over h2,
# percent-decoded `:path`, and `error_page` for an invalid `:authority`.
h2_absolute_redirect.t
# `add_trailer` over h2: HEADERS, DATA without END_STREAM, then a trailer HEADERS
# (split into CONTINUATION when the block exceeds the frame size). Also pins that
# the hop-by-hop headers HTTP/2 forbids are not forwarded.
h2_trailers.t
# Location-scope `server_tokens`, in the `Server` line and the builtin error
# body, over both protocols.
h2_server_tokens.t
server_tokens.t

# ---- request body not buffered ----
# `proxy_request_buffering off`: the upstream connection is made before the
# client's body is read, and the body is forwarded as it arrives.
proxy_request_buffering.t

# ---- slice filter / fastcgi cache ----
# `slice` (fixed-size ranged subrequests over proxy and fastcgi, assembled into
# one response, each slice its own cache entry) and `fastcgi_cache`.
slice.t
fastcgi_cache.t

# ---- upstream keepalive pool ----
# The pool itself: buffered, unbuffered and in-memory (SSI subrequest) paths all
# reuse it; a chunked reply is a message boundary like a Content-Length one; and
# a client that cannot be sent chunked framing (HTTP/1.0) gets a dechunked body.
proxy_keepalive.t
proxy_chunked.t

# ---- stream upstream balancers ----
# `max_conns` (per-peer cap, two-pool primary/backup walk), `hash` (both the
# folded Cache::Memcached expression and the consistent ring), `least_conn`
# with its smooth-WRR tie-break, and `random two least_conn`.
# `stream_upstream.t`: the same balancers plus `max_fails`/`fail_timeout`
# recovery, the single-peer rule, and `$upstream_addr` /
# `$upstream_bytes_sent` / `$upstream_bytes_received` as per-attempt lists.
stream_upstream.t
stream_upstream_max_conns.t
stream_upstream_hash.t
stream_upstream_least_conn.t
stream_upstream_random.t

# ---- charset filter ----
# `charset` / `source_charset` / `charset_types` / `override_charset` /
# `charset_map`, including the reverse direction of a declared map (one
# `charset_map B A` serves B->A and A->B, like nginx's two table registrations).
charset.t
# `gzip_static`: a sibling `.gz` replaces the entity, with Content-Type still
# taken from the original name -- and the combination with charset, where a
# response that *would* need recoding is left completely alone.
charset_gzip_static.t

# ---- index ----
# `index` as a complex value, its internal redirect (`$uri` changes), absolute
# index names, the redirect-cycle 500, and `log_not_found`.
index.t

# ---- headers ----
# `add_header` with `always`, the special-header names (`ETag`/`Last-Modified`
# are *set*, so an empty value clears them), and `expires` in every form
# including a complex value (`expires $arg_e`).
headers.t

# ---- request body ----
# `$request_body` / `$request_body_file` (the body has to be in memory before
# dispatch, because the variable is read by `add_header` after the upstream
# answered), `client_body_buffer_size` / `client_body_in_file_only` spilling,
# `client_max_body_size` for chunked bodies, and chunk-framing validation.
body.t
body_chunked.t

# ---- resolver ----
# A + AAAA in parallel, CNAME following with nginx's `process_a` state machine,
# TTL/`valid=` caching, per-name server rotation, resend, and TCP fallback on a
# truncated answer.
http_resolver.t
http_resolver_aaaa.t

# ---- autoindex ----
# `autoindex_format html|json|jsonp|xml`. nginx's JSON/XML bodies are compared
# byte for byte, including the fact that entry names are emitted as raw UTF-8
# bytes and only `\\`, `"` and control characters are escaped.
autoindex_format.t

# ---- PROXY protocol ----
# `listen ... proxy_protocol` (v1 text and v2 binary with TLVs),
# `$proxy_protocol_*`, `real_ip_header proxy_protocol`, and `stream`'s
# `proxy_protocol on`. `allow`/`deny` and the access log follow the rewritten

proxy_protocol2_tlv.t
proxy_protocol_ipv6.t

# ---- dav ----
